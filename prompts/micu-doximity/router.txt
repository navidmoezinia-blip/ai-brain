IN UPLOADED FILES: .micuhp = DOX_MICU_HP; .micucheck = DOX_MICU_CHECK; .micupres = DOX_MICU_PRESENT; .micuhandoff = DOX_MICU_HANDOFF; .micuconsult = DOX_MICU_CONSULT.
